cmake_minimum_required(VERSION 3.22)
project(OneTunnelNativeWfpGuard VERSION 0.1.0 LANGUAGES CXX)
if (NOT WIN32)
  message(FATAL_ERROR "This target requires Windows")
endif()
add_executable(onetunnel-wfp-guard wfp_guard.cpp)
target_compile_features(onetunnel-wfp-guard PRIVATE cxx_std_17)
target_compile_definitions(onetunnel-wfp-guard PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(onetunnel-wfp-guard PRIVATE Fwpuclnt Rpcrt4)
if (MSVC)
  target_compile_options(onetunnel-wfp-guard PRIVATE /W4 /EHsc)
endif()

add_executable(onetunnel-transparent-relay transparent_relay.cpp)
target_compile_features(onetunnel-transparent-relay PRIVATE cxx_std_17)
target_compile_definitions(onetunnel-transparent-relay PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(onetunnel-transparent-relay PRIVATE Ws2_32)
if (MSVC)
  target_compile_options(onetunnel-transparent-relay PRIVATE /W4 /EHsc)
endif()

add_executable(onetunnel-wfp-control wfp_control.cpp)
target_compile_features(onetunnel-wfp-control PRIVATE cxx_std_17)
target_compile_definitions(onetunnel-wfp-control PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(onetunnel-wfp-control PRIVATE Fwpuclnt Rpcrt4)
if (MSVC)
  target_compile_options(onetunnel-wfp-control PRIVATE /W4 /EHsc)
endif()
